```python
"""Walk-forward CLV backtest math -- ML-02.

CLV staked odds MUST use opening odds x (1 - slippage_pct), NOT closing odds.
Retrofitting closing-odds CLV later invalidates every earlier backtest result.
"""

from __future__ import annotations

SLIPPAGE_PCT: float = 0.015  # 1.5% -- mid-range of 1-2% spec (ML-02)


def apply_slippage(opening_odds: float) -> float:
    """Bet-taker loses SLIPPAGE_PCT of odds edge to market movement."""
    return opening_odds * (1.0 - SLIPPAGE_PCT)
```

**Pure-function + terse docstring pattern** (lines 17-23):
```python
def compute_clv(staked_odds: float, pinnacle_closing: float) -> float:
    """CLV percentage vs Pinnacle closing (ML-02 / CLV-02).

    CLV% = (staked_odds / pinnacle_closing - 1) * 100.
    Positive -> the bet was taken at a better price than the closing line.
    """
    return (staked_odds / pinnacle_closing - 1.0) * 100.0
```

```python
import numpy as np   # NEW top-level import

EDGE_THRESHOLD_PCT: float = 0.05  # D-10 — Phase 3 pick engine imports this.


def simulate_pick(
    model_probs: np.ndarray,         # shape (3,): home/draw/away probabilities
    opening_odds: np.ndarray,        # shape (3,): Betano decimal odds; NaN allowed
    threshold: float = EDGE_THRESHOLD_PCT,
) -> int | None:
    """Return outcome index (0/1/2) with max edge if edge >= threshold, else None.

    Returns None when:
      - any opening_odds value is NaN (Betano coverage gap; D-03), OR
      - max edge < threshold.
    """
    if np.any(np.isnan(opening_odds)):
        return None
    edge = model_probs * opening_odds - 1.0
    if float(np.max(edge)) < threshold:
        return None
    return int(np.argmax(edge))
```

```python
import sys

import psycopg
import structlog

from bip.core.settings import Settings

def derive_db_url(supabase_url: str, db_password: str) -> str:
    subdomain = supabase_url.removeprefix("https://").removesuffix(".supabase.co")
    return (
        f"postgresql://postgres:{db_password}"
        f"@db.{subdomain}.supabase.co:5432/postgres"
    )


def main() -> int:
    settings = Settings()
    db_password = os.environ.get("SUPABASE_DB_PASSWORD")
    if not db_password:
        print("FAIL: SUPABASE_DB_PASSWORD not set in environment")
        return 1
    db_url = derive_db_url(settings.supabase_url, db_password)
    with psycopg.connect(db_url) as conn, conn.cursor() as cur:
        cur.execute(
            "SELECT column_name, data_type FROM information_schema.columns "
            "WHERE table_name=%s AND column_name=%s",
            ("predictions", "is_shadow"),
        )
        if not cur.fetchall():
            print("FAIL: is_shadow column not found on predictions table")
            return 1
        cur.execute(
            "SELECT indexname FROM pg_indexes WHERE tablename=%s AND indexname=%s",
            ("predictions", "idx_predictions_is_shadow"),
        )
        if not cur.fetchall():
            print("FAIL: idx_predictions_is_shadow not found")
            return 1
    logger.info("migration_003_verified")
    print("OK: migration 003 applied — is_shadow column + index present")
    return 0


if __name__ == "__main__":
```
