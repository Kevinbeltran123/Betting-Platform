```python
"""Structured logging setup using structlog."""

import sys
import structlog


def configure_logging(log_level: str = "INFO") -> None:
    """Configure structlog for JSON in production, pretty in terminal."""
    shared_processors = [
        structlog.stdlib.add_log_level,
        structlog.stdlib.add_logger_name,
        structlog.processors.TimeStamper(fmt="iso", utc=True),
        structlog.processors.StackInfoRenderer(),
    ]
    if sys.stderr.isatty():
        processors = shared_processors + [structlog.dev.ConsoleRenderer()]
    else:
        processors = shared_processors + [
            structlog.processors.dict_tracebacks,
            structlog.processors.JSONRenderer(),
        ]

    structlog.configure(
        processors=processors,
        wrapper_class=structlog.make_filtering_bound_logger(
            getattr(__import__("logging"), log_level)
        ),
        context_class=dict,
        logger_factory=structlog.PrintLoggerFactory(),
        cache_logger_on_first_use=True,
    )
```

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Information Disclosure | `Settings` object | mitigate | structlog must never log the Settings object — log individual non-secret fields only (e.g., `log_level`, `parquet_base_path`). Never `logger.info("settings", settings=settings)`. |
| T-03-02 | Information Disclosure | `supabase_key` in `Settings` | mitigate | `pydantic-settings` loads from `.env` (in `.gitignore`); env var is `SUPABASE_KEY`. The key never appears in source code. |
| T-03-03 | Tampering | `market: str` field in models | accept | Market string is validated at application layer (YAML config); storage layer accepts any string. Invalid market strings fail at EV engine (Phase 3), not storage. |
