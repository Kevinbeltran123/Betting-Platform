```python
"""Send-only Telegram bot wrapper.

D-13: Lives in the same asyncio loop as PipelineOrchestrator (in-process). NO polling —
single private channel, outbound alerts only. AIORateLimiter handles flood control
(Pitfall 2 + Risks 2 — requires the [rate-limiter] extra in pyproject.toml).

PATTERNS.md drift risks:
  #7: ALWAYS call initialize() then start() — initialize alone doesn't make bot.send_message callable.
  #8: NEVER call application.updater.start_polling() — burns API quota for no benefit.
  #13: telegram_channel_id stays str in Settings; cast to int at the send call site.
"""

from __future__ import annotations

import structlog
from telegram import Bot
from telegram.constants import ParseMode
from telegram.ext import AIORateLimiter, Application, ApplicationBuilder

from bip.core.errors import TelegramError

logger = structlog.get_logger(__name__)


class TelegramBot:
    """Outbound-only Telegram bot owned by PipelineOrchestrator for the process lifetime."""

    def __init__(self, token: str, channel_id: str) -> None:
        if not token:
            raise TelegramError("telegram_bot_token is empty — set TELEGRAM_BOT_TOKEN in .env")
        if not channel_id:
            raise TelegramError("telegram_channel_id is empty — set TELEGRAM_CHANNEL_ID in .env")

        # PATTERNS.md drift risk #13: keep channel_id as str; cast to int at send site.
        self._channel_id = channel_id
        self._app: Application = (
            ApplicationBuilder()
            .token(token)
            .rate_limiter(AIORateLimiter(max_retries=3))   # Pitfall 2: auto-handle RetryAfter
            .build()
        )

    async def start(self) -> None:
        """Initialize + start. NO updater.start_polling() — send-only (drift risk #8)."""
        await self._app.initialize()
        await self._app.start()
        logger.info("telegram_bot_started", channel_id=self._channel_id)

    async def shutdown(self) -> None:
        """Graceful shutdown — call from PipelineOrchestrator.shutdown."""
        await self._app.stop()
        await self._app.shutdown()
        logger.info("telegram_bot_stopped")

    @property
    def bot(self) -> Bot:
        return self._app.bot

    async def send_html(self, text: str) -> None:
        """Send one HTML-formatted message to the private channel (D-12)."""
        await self.bot.send_message(
            chat_id=int(self._channel_id),     # drift risk #13: cast at use-site
            text=text,
            parse_mode=ParseMode.HTML,
            disable_web_page_preview=True,
        )
        logger.info("telegram_send_success", channel_id=self._channel_id, length=len(text))
```

```python
"""Telegram pick sender — renders Pick rows through Jinja2 and pushes to TelegramBot.

D-12: parse_mode='HTML' (handled by TelegramBot.send_html).
D-14: FLAG verdict → WARNING marker (template branch).
D-15: bullets sourced from claude_summary, split on ' * ', capped at 3.
"""

from __future__ import annotations

from typing import Any

import structlog
from jinja2 import Environment, PackageLoader, select_autoescape

from bip.core.storage.models import Pick
from bip.core.telegram.bot import TelegramBot

logger = structlog.get_logger(__name__)

_env = Environment(
    loader=PackageLoader("bip.core.telegram", "templates"),
    autoescape=select_autoescape(["html"]),     # T-3-XSS: auto-escape & < >
    trim_blocks=True,
    lstrip_blocks=True,
)


def _build_context(pick: Pick, home_team: str, away_team: str, model_version: str,
                    reason_code: str | None) -> dict[str, Any]:
    """Build the dict passed as `pick` to the template.

    Pick model carries data; presentation fields (fixture display, reason_code from
    validator output, model_version from prediction) come from the call site.
    """
    return {
        "claude_validation": pick.claude_validation,
        "fixture": f"{home_team} vs {away_team}",
        "market": pick.market,
        "selection": pick.selection,
        "best_odds": pick.best_odds,
        "edge": pick.edge,
        "suggested_stake": pick.suggested_stake,
        "model_probability": pick.model_probability,
        "bookmaker": pick.bookmaker,
        "model_version": model_version,
        "reason_code": reason_code or "",
    }


def render_pick(
    pick: Pick,
    home_team: str,
    away_team: str,
    model_version: str,
    reason_code: str | None = None,
) -> str:
    """Render a Pick to HTML.

    D-15: claude_summary bullets are split on ' * ' separator (literal: space-asterisk-space).
    Capped at 3. Empty claude_summary → no bullets section.
    """
    raw = pick.claude_summary or ""
    bullets = [b.strip() for b in raw.split(" * ") if b.strip()]
    ctx = _build_context(pick, home_team, away_team, model_version, reason_code)
    return _env.get_template("pick.html").render(pick=ctx, summary_bullets=bullets)


class TelegramSender:
    """Pushes rendered picks through TelegramBot. One message per pick (D-12)."""

    def __init__(self, bot: TelegramBot) -> None:
        self._bot = bot

    async def send_pick(
        self,
        pick: Pick,
        home_team: str,
        away_team: str,
        model_version: str,
        reason_code: str | None = None,
    ) -> None:
        """D-12: one Telegram message per qualifying pick — never batched.

        REJECT verdicts MUST be filtered at the engine layer; sender does not enforce.
        """
        text = render_pick(pick, home_team, away_team, model_version, reason_code)
        await self._bot.send_html(text)
        logger.info(
            "telegram_pick_sent",
            fixture_id=pick.fixture_id,
            market=pick.market,
            verdict=pick.claude_validation,
            length=len(text),
        )
```
